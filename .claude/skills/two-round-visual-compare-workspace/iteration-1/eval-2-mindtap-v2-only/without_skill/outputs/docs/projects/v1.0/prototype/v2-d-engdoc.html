## 代码片段

# 从 hex + RGB 推断近似色温
def kelvin_from_rgb(r, g, b):
    # McCamy 近似
    x = (r - b) / (r + g + b + 1e-6)
    y = (g - 0.5) / (r + 1e-6)
    return round(4000 + x * 3500 + y * 1200)

print(kelvin_from_rgb(58, 53, 82))  # → 6702
